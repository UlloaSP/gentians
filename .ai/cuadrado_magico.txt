% -----------------------------------------------------------------------------
% cuadrado_magico.txt
%
% Tarea para Gentians: aprender invalid/1 para cuadrado magico 3x3.
%
% Las reglas que deberia inferir estan comentadas abajo. El background deja
% disponible el dominio, un cuadrado valido y varios cuadrados invalidos.
% -----------------------------------------------------------------------------

% -----------------------------------------------------------------------------
% Reglas objetivo esperadas, comentadas.
% -----------------------------------------------------------------------------

% Una celda no puede tener dos valores:
% invalid(B) :- board(B), row(R), col(C), val(V1), val(V2),
%    cell(B,R,C,V1), cell(B,R,C,V2), V1 != V2.

% Un valor no puede aparecer dos veces:
% invalid(B) :- board(B), val(V), row(R1), col(C1), row(R2), col(C2),
%    cell(B,R1,C1,V), cell(B,R2,C2,V),
%    R1 != R2, C1 != C2.

% Cada fila suma magic:
% invalid(B) :- board(B), row(R), magic(M),
%    M != #sum { V,C : cell(B,R,C,V) }.

% Cada columna suma magic:
% invalid(B) :- board(B), col(C), magic(M),
%    M != #sum { V,R : cell(B,R,C,V) }.

% Diagonal principal suma magic:
% invalid(B) :- board(B), magic(M),
%    M != #sum { V,R,C : cell(B,R,C,V), diag_main(R,C) }.

% Diagonal secundaria suma magic:
% invalid(B) :- board(B), magic(M),
%    M != #sum { V,R,C : cell(B,R,C,V), diag_anti(R,C) }.
%
% Para usarlo como constraint ASP despues:
% :- invalid(B).

% -----------------------------------------------------------------------------
% Background: dominio fijo 3x3.
% -----------------------------------------------------------------------------

row(1..3).
col(1..3).
val(1..9).
magic(15).

diag_main(1,1).
diag_main(2,2).
diag_main(3,3).

diag_anti(1,3).
diag_anti(2,2).
diag_anti(3,1).

% -----------------------------------------------------------------------------
% Boards de entrenamiento.
% b1 es Lo Shu valido.
% b2 rompe fila/columna/diagonal.
% b3 repite valores.
% b4 rompe diagonales.
% -----------------------------------------------------------------------------

board(b1).
cell(b1,1,1,8). cell(b1,1,2,1). cell(b1,1,3,6).
cell(b1,2,1,3). cell(b1,2,2,5). cell(b1,2,3,7).
cell(b1,3,1,4). cell(b1,3,2,9). cell(b1,3,3,2).

board(b2).
cell(b2,1,1,1). cell(b2,1,2,2). cell(b2,1,3,3).
cell(b2,2,1,4). cell(b2,2,2,5). cell(b2,2,3,6).
cell(b2,3,1,7). cell(b2,3,2,8). cell(b2,3,3,9).

board(b3).
cell(b3,1,1,8). cell(b3,1,2,1). cell(b3,1,3,6).
cell(b3,2,1,3). cell(b3,2,2,5). cell(b3,2,3,7).
cell(b3,3,1,4). cell(b3,3,2,4). cell(b3,3,3,2).

board(b4).
cell(b4,1,1,8). cell(b4,1,2,1). cell(b4,1,3,6).
cell(b4,2,1,7). cell(b4,2,2,5). cell(b4,2,3,3).
cell(b4,3,1,4). cell(b4,3,2,9). cell(b4,3,3,2).

% -----------------------------------------------------------------------------
% Ejemplos.
% Gentians debe aprender invalid/1:
% - b1 no es invalido.
% - b2, b3 y b4 son invalidos.
% -----------------------------------------------------------------------------

#pos({invalid(b2)},{}).
#pos({invalid(b3)},{}).
#pos({invalid(b4)},{}).

#neg({invalid(b1)},{}).

% -----------------------------------------------------------------------------
% Bias.
%
% Para agregados usar CLI, no archivo:
%   --aggregates "sum(cell/4)" "count(cell/4)"
%
% Para comparaciones:
%   --comparison neq eq
% -----------------------------------------------------------------------------

#modeh(1,invalid,1).

#modeb(1,board,1,positive).
#modeb(1,row,1,positive).
#modeb(1,col,1,positive).
#modeb(1,val,1,positive).
#modeb(1,magic,1,positive).
#modeb(2,cell,4,positive).
#modeb(1,diag_main,2,positive).
#modeb(1,diag_anti,2,positive).
