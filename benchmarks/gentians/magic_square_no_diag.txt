% 1 { x(Row, Col, N) : size(Row), size(Col) } 1 :- val(N).
% % To learn:

% :- sum_col(C0,SC0), sum_col(C1,SC1), C0 != C1, SC0 != SC1.
% :- sum_row(R0,SR0), sum_row(R1,SR1), R0 != R1, SR0 != SR1.
